-- =========================================================
-- CAR DEALERSHIP SAMPLE DATABASE
-- For use in DB Fiddle (PostgreSQL/MySQL compatible)
-- =========================================================

-- ---------- SCHEMA ----------

CREATE TABLE dealers (
    dealer_id    INTEGER PRIMARY KEY,
    dealer_name  VARCHAR(100) NOT NULL,
    city         VARCHAR(50),
    state        VARCHAR(50),
    phone        VARCHAR(20)
);

CREATE TABLE brands (
    brand_id    INTEGER PRIMARY KEY,
    brand_name  VARCHAR(50) NOT NULL
);

CREATE TABLE models (
    model_id    INTEGER PRIMARY KEY,
    brand_id    INTEGER REFERENCES brands(brand_id),
    model_name  VARCHAR(50) NOT NULL,
    category    VARCHAR(30)
);

CREATE TABLE employees (
    employee_id  INTEGER PRIMARY KEY,
    dealer_id    INTEGER REFERENCES dealers(dealer_id),
    first_name   VARCHAR(50) NOT NULL,
    last_name    VARCHAR(50) NOT NULL,
    role         VARCHAR(20) NOT NULL,  -- 'Manager' or 'Seller'
    hire_date    DATE
);

CREATE TABLE cars (
    car_id       INTEGER PRIMARY KEY,
    dealer_id    INTEGER REFERENCES dealers(dealer_id),
    model_id     INTEGER REFERENCES models(model_id),
    year         INTEGER NOT NULL,
    color        VARCHAR(30),
    vin          VARCHAR(20) UNIQUE,
    price        DECIMAL(10,2) NOT NULL,
    status       VARCHAR(20) NOT NULL  -- 'Available' or 'Sold'
);

CREATE TABLE customers (
    customer_id  INTEGER PRIMARY KEY,
    first_name   VARCHAR(50) NOT NULL,
    last_name    VARCHAR(50) NOT NULL,
    email        VARCHAR(100),
    phone        VARCHAR(20),
    city         VARCHAR(50)
);

CREATE TABLE sales (
    sale_id      INTEGER PRIMARY KEY,
    car_id       INTEGER REFERENCES cars(car_id),
    customer_id  INTEGER REFERENCES customers(customer_id),
    employee_id  INTEGER REFERENCES employees(employee_id),
    sale_date    DATE NOT NULL,
    sale_price   DECIMAL(10,2) NOT NULL
);
